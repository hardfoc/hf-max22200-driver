224inline uint32_t getChopFreqKhz(bool master_clock_80khz, ChopFreq cf) {
225 switch (cf) {
226 case ChopFreq::FMAIN_DIV4: return master_clock_80khz ? 20u : 25u;
227 case ChopFreq::FMAIN_DIV3: return master_clock_80khz ? 26u : 33u;
228 case ChopFreq::FMAIN_DIV2: return master_clock_80khz ? 40u : 50u;
229 case ChopFreq::FMAIN: return master_clock_80khz ? 80u : 100u;
230 default: return 25u;
231 }

240inline uint8_t currentMaToRaw(uint32_t full_scale_current_ma, uint32_t ma) {
241 if (full_scale_current_ma == 0) return 0;
242 if (ma >= full_scale_current_ma) return 127;
243 uint32_t raw = (ma * 127u + full_scale_current_ma / 2u) / full_scale_current_ma;
244 return raw > 127u ? 127u : static_cast<uint8_t>(raw);
245}

254inline uint8_t hitTimeMsToRaw(float ms, bool master_clock_80khz, ChopFreq cf) {
255 uint32_t fchop_khz = getChopFreqKhz(master_clock_80khz, cf);
256 if (ms < 0.0f || ms >= 1000000.0f) return 255;
257 if (ms == 0.0f) return 0;
258 float fchop_hz = static_cast<float>(fchop_khz) * 1000.0f;
259 uint32_t raw = static_cast<uint32_t>((ms / 1000.0f) * fchop_hz / 40.0f + 0.5f);
260 if (raw > 254u) return 255;
261 if (raw == 0u) return 1;
262 return static_cast<uint8_t>(raw);
263}

455 uint32_t toRegister(uint32_t board_ifs_ma = 0, bool master_clock_80khz = false) const {
456 // Effective IFS for CDR: HFS=1 halves the scale (datasheet KFS 7.5k vs 15k)
457 const uint32_t ifs_ma = (drive_mode == DriveMode::CDR && half_full_scale && board_ifs_ma >= 2u)
458 ? (board_ifs_ma / 2u)
459 : board_ifs_ma;
460
461 // Convert hit_setpoint to raw 7-bit
462 uint8_t hit_raw;
463 if (drive_mode == DriveMode::CDR) {
464 // CDR: hit_setpoint is in mA (relative to effective IFS for this channel)
465 if (ifs_ma == 0) {
466 hit_raw = 0; // Invalid IFS
467 } else if (hit_setpoint >= static_cast<float>(ifs_ma)) {
468 hit_raw = 127;
469 } else {
470 uint32_t ma = static_cast<uint32_t>(hit_setpoint + 0.5f);
471 hit_raw = currentMaToRaw(ifs_ma, ma);
472 }
473 } else {
474 // VDR: hit_setpoint is duty percent (0-100)
475 if (hit_setpoint <= 0.0f) {
476 hit_raw = 0;
477 } else if (hit_setpoint >= 100.0f) {
478 hit_raw = 127;
479 } else {
480 uint32_t r = static_cast<uint32_t>((hit_setpoint / 100.0f) * 127.0f + 0.5f);
481 hit_raw = (r > 127u) ? 127u : static_cast<uint8_t>(r);
482 }
483 }
484
485 // Convert hold_setpoint to raw 7-bit
486 uint8_t hold_raw;
487 if (drive_mode == DriveMode::CDR) {
488 // CDR: hold_setpoint is in mA
489 if (ifs_ma == 0) {
490 hold_raw = 0;
491 } else if (hold_setpoint >= static_cast<float>(ifs_ma)) {
492 hold_raw = 127;
493 } else {
494 uint32_t ma = static_cast<uint32_t>(hold_setpoint + 0.5f);
495 hold_raw = currentMaToRaw(ifs_ma, ma);
496 }
497 } else {
498 // VDR: hold_setpoint is duty percent (0-100)
499 if (hold_setpoint <= 0.0f) {
500 hold_raw = 0;
501 } else if (hold_setpoint >= 100.0f) {
502 hold_raw = 127;
503 } else {
504 uint32_t r = static_cast<uint32_t>((hold_setpoint / 100.0f) * 127.0f + 0.5f);
505 hold_raw = (r > 127u) ? 127u : static_cast<uint8_t>(r);
506 }
507 }
508
509 // Convert hit_time_ms to raw 8-bit (master_clock_80khz from parameter, not stored on config)
510 uint8_t hit_time_raw = hitTimeMsToRaw(hit_time_ms, master_clock_80khz, chop_freq);
511
512 // Build register value
513 uint32_t val = 0;
514 if (half_full_scale) val |= CfgChReg::HFS_BIT;
515 val |= (static_cast<uint32_t>(hold_raw & 0x7F) << CfgChReg::HOLD_SHIFT);
516 if (trigger_from_pin) val |= CfgChReg::TRGNSPI_BIT;
517 val |= (static_cast<uint32_t>(hit_raw & 0x7F) << CfgChReg::HIT_SHIFT);
518 val |= (static_cast<uint32_t>(hit_time_raw) << CfgChReg::HITT_SHIFT);
519 if (drive_mode == DriveMode::VDR) val |= CfgChReg::VDRNCDR_BIT;
520 if (side_mode == SideMode::HIGH_SIDE) val |= CfgChReg::HSNLS_BIT;
521 val |= (static_cast<uint32_t>(chop_freq) << CfgChReg::FREQ_CFG_SHIFT);
522 if (slew_rate_control_enabled) val |= CfgChReg::SRC_BIT;
523 if (open_load_detection_enabled) val |= CfgChReg::OL_EN_BIT;
524 if (plunger_movement_detection_enabled) val |= CfgChReg::DPM_EN_BIT;
525 if (hit_current_check_enabled) val |= CfgChReg::HHF_EN_BIT;
526 return val;
527 }

538 void fromRegister(uint32_t val, uint32_t board_ifs_ma, bool master_clock_80khz_param) {
539 // Parse register fields
540 half_full_scale = (val & CfgChReg::HFS_BIT) != 0;
541 uint8_t hold_raw = static_cast<uint8_t>((val >> CfgChReg::HOLD_SHIFT) & 0x7F);
542 trigger_from_pin = (val & CfgChReg::TRGNSPI_BIT) != 0;
543 uint8_t hit_raw = static_cast<uint8_t>((val >> CfgChReg::HIT_SHIFT) & 0x7F);
544 uint8_t hit_time_raw = static_cast<uint8_t>((val >> CfgChReg::HITT_SHIFT) & 0xFF);
545 drive_mode = (val & CfgChReg::VDRNCDR_BIT) ? DriveMode::VDR : DriveMode::CDR;
546 side_mode = (val & CfgChReg::HSNLS_BIT) ? SideMode::HIGH_SIDE : SideMode::LOW_SIDE;
547 chop_freq = static_cast<ChopFreq>((val >> CfgChReg::FREQ_CFG_SHIFT) & 0x03);
548 slew_rate_control_enabled = (val & CfgChReg::SRC_BIT) != 0;
549 open_load_detection_enabled = (val & CfgChReg::OL_EN_BIT) != 0;
550 plunger_movement_detection_enabled = (val & CfgChReg::DPM_EN_BIT) != 0;
551 hit_current_check_enabled = (val & CfgChReg::HHF_EN_BIT) != 0;
552
553 // Convert raw → user units (master_clock_80khz_param used for hit_time only; not stored on config)
554 if (drive_mode == DriveMode::CDR) {
555 // CDR: raw → mA (use effective IFS when HFS=1)
556 const uint32_t effective_ifs = half_full_scale && board_ifs_ma >= 2u
557 ? (board_ifs_ma / 2u)
558 : board_ifs_ma;
559 if (effective_ifs > 0) {
560 hit_setpoint = (static_cast<float>(hit_raw) / 127.0f) * static_cast<float>(effective_ifs);
561 hold_setpoint = (static_cast<float>(hold_raw) / 127.0f) * static_cast<float>(effective_ifs);
562 } else {
563 hit_setpoint = 0.0f;
564 hold_setpoint = 0.0f;
565 }
566 } else {
567 // VDR: raw → duty percent
568 hit_setpoint = (static_cast<float>(hit_raw) / 127.0f) * 100.0f;
569 hold_setpoint = (static_cast<float>(hold_raw) / 127.0f) * 100.0f;
570 }
571
572 // Convert hit_time raw → ms
573 if (hit_time_raw == 0) {
574 hit_time_ms = 0.0f;
575 } else if (hit_time_raw == 255) {
576 hit_time_ms = -1.0f; // Continuous (infinite)
577 } else {
578 uint32_t fchop_khz = getChopFreqKhz(master_clock_80khz_param, chop_freq);
579 float fchop_hz = static_cast<float>(fchop_khz) * 1000.0f;
580 hit_time_ms = (static_cast<float>(hit_time_raw) * 40.0f / fchop_hz) * 1000.0f;
581 }
582 }

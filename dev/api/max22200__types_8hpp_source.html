224inline uint32_t getChopFreqKhz(bool master_clock_80khz, ChopFreq cf) {

257inline uint8_t currentMaToRaw(uint32_t full_scale_current_ma, uint32_t ma) {
258 if (full_scale_current_ma == 0) return 0;
259 if (ma >= full_scale_current_ma) return 127;
260 uint32_t raw = (ma * 127u + full_scale_current_ma / 2u) / full_scale_current_ma;
261 return raw > 127u ? 127u : static_cast<uint8_t>(raw);
262}

296inline uint8_t hitTimeMsToRaw(float ms, bool master_clock_80khz, ChopFreq cf) {
297 uint32_t fchop_khz = getChopFreqKhz(master_clock_80khz, cf);
298 if (ms < 0.0f || ms >= 1000000.0f) return 255;
299 if (ms == 0.0f) return 0;
300 float fchop_hz = static_cast<float>(fchop_khz) * 1000.0f;
301 uint32_t raw = static_cast<uint32_t>((ms / 1000.0f) * fchop_hz / 40.0f + 0.5f);
302 if (raw > 254u) return 255;
303 if (raw == 0u) return 1;
304 return static_cast<uint8_t>(raw);
305}

497 uint32_t toRegister(uint32_t board_ifs_ma = 0, bool master_clock_80khz = false) const {
498 // Effective IFS for CDR: HFS=1 halves the scale (datasheet KFS 7.5k vs 15k)
499 const uint32_t ifs_ma = (drive_mode == DriveMode::CDR && half_full_scale && board_ifs_ma >= 2u)
500 ? (board_ifs_ma / 2u)
501 : board_ifs_ma;
502
503 // Convert hit_setpoint to raw 7-bit
504 uint8_t hit_raw;
505 if (drive_mode == DriveMode::CDR) {
506 // CDR: hit_setpoint is in mA (relative to effective IFS for this channel)
507 if (ifs_ma == 0) {
508 hit_raw = 0; // Invalid IFS
509 } else if (hit_setpoint >= static_cast<float>(ifs_ma)) {
510 hit_raw = 127;
511 } else {
512 uint32_t ma = static_cast<uint32_t>(hit_setpoint + 0.5f);
513 hit_raw = currentMaToRaw(ifs_ma, ma);
514 }
515 } else {
516 // VDR: hit_setpoint is duty percent (0-100)
517 hit_raw = dutyPercentToRaw(hit_setpoint);
518 }
519
520 // Convert hold_setpoint to raw 7-bit
521 uint8_t hold_raw;
522 if (drive_mode == DriveMode::CDR) {
523 // CDR: hold_setpoint is in mA
524 if (ifs_ma == 0) {
525 hold_raw = 0;
526 } else if (hold_setpoint >= static_cast<float>(ifs_ma)) {
527 hold_raw = 127;
528 } else {
529 uint32_t ma = static_cast<uint32_t>(hold_setpoint + 0.5f);
530 hold_raw = currentMaToRaw(ifs_ma, ma);
531 }
532 } else {
533 // VDR: hold_setpoint is duty percent (0-100)
534 hold_raw = dutyPercentToRaw(hold_setpoint);
535 }
536
537 // Convert hit_time_ms to raw 8-bit (master_clock_80khz from parameter, not stored on config)
538 uint8_t hit_time_raw = hitTimeMsToRaw(hit_time_ms, master_clock_80khz, chop_freq);
539
540 // Build register value
541 uint32_t val = 0;
542 if (half_full_scale) val |= CfgChReg::HFS_BIT;
543 val |= (static_cast<uint32_t>(hold_raw & 0x7F) << CfgChReg::HOLD_SHIFT);
544 if (trigger_from_pin) val |= CfgChReg::TRGNSPI_BIT;
545 val |= (static_cast<uint32_t>(hit_raw & 0x7F) << CfgChReg::HIT_SHIFT);
546 val |= (static_cast<uint32_t>(hit_time_raw) << CfgChReg::HITT_SHIFT);
547 if (drive_mode == DriveMode::VDR) val |= CfgChReg::VDRNCDR_BIT;
548 if (side_mode == SideMode::HIGH_SIDE) val |= CfgChReg::HSNLS_BIT;
549 val |= (static_cast<uint32_t>(chop_freq) << CfgChReg::FREQ_CFG_SHIFT);
550 if (slew_rate_control_enabled) val |= CfgChReg::SRC_BIT;
551 if (open_load_detection_enabled) val |= CfgChReg::OL_EN_BIT;
552 if (plunger_movement_detection_enabled) val |= CfgChReg::DPM_EN_BIT;
553 if (hit_current_check_enabled) val |= CfgChReg::HHF_EN_BIT;
554 return val;
555 }

566 void fromRegister(uint32_t val, uint32_t board_ifs_ma, bool master_clock_80khz_param) {
567 // Parse register fields
568 half_full_scale = (val & CfgChReg::HFS_BIT) != 0;
569 uint8_t hold_raw = static_cast<uint8_t>((val >> CfgChReg::HOLD_SHIFT) & 0x7F);
570 trigger_from_pin = (val & CfgChReg::TRGNSPI_BIT) != 0;
571 uint8_t hit_raw = static_cast<uint8_t>((val >> CfgChReg::HIT_SHIFT) & 0x7F);
572 uint8_t hit_time_raw = static_cast<uint8_t>((val >> CfgChReg::HITT_SHIFT) & 0xFF);
573 drive_mode = (val & CfgChReg::VDRNCDR_BIT) ? DriveMode::VDR : DriveMode::CDR;
574 side_mode = (val & CfgChReg::HSNLS_BIT) ? SideMode::HIGH_SIDE : SideMode::LOW_SIDE;
575 chop_freq = static_cast<ChopFreq>((val >> CfgChReg::FREQ_CFG_SHIFT) & 0x03);
576 slew_rate_control_enabled = (val & CfgChReg::SRC_BIT) != 0;
577 open_load_detection_enabled = (val & CfgChReg::OL_EN_BIT) != 0;
578 plunger_movement_detection_enabled = (val & CfgChReg::DPM_EN_BIT) != 0;
579 hit_current_check_enabled = (val & CfgChReg::HHF_EN_BIT) != 0;
580
581 // Convert raw → user units (master_clock_80khz_param used for hit_time only; not stored on config)
582 if (drive_mode == DriveMode::CDR) {
583 // CDR: raw → mA (use effective IFS when HFS=1)
584 const uint32_t effective_ifs = half_full_scale && board_ifs_ma >= 2u
585 ? (board_ifs_ma / 2u)
586 : board_ifs_ma;
587 if (effective_ifs > 0) {
588 hit_setpoint = (static_cast<float>(hit_raw) / 127.0f) * static_cast<float>(effective_ifs);
589 hold_setpoint = (static_cast<float>(hold_raw) / 127.0f) * static_cast<float>(effective_ifs);
590 } else {
591 hit_setpoint = 0.0f;
592 hold_setpoint = 0.0f;
593 }
594 } else {
595 // VDR: the field is the duty cycle in percent (datasheet Table 3); 100 and above is 100 %
596 hit_setpoint = dutyRawToPercent(hit_raw);
597 hold_setpoint = dutyRawToPercent(hold_raw);
598 }
599
600 // Convert hit_time raw → ms
601 if (hit_time_raw == 0) {
602 hit_time_ms = 0.0f;
603 } else if (hit_time_raw == 255) {
604 hit_time_ms = -1.0f; // Continuous (infinite)
605 } else {
606 uint32_t fchop_khz = getChopFreqKhz(master_clock_80khz_param, chop_freq);
607 float fchop_hz = static_cast<float>(fchop_khz) * 1000.0f;
608 hit_time_ms = (static_cast<float>(hit_time_raw) * 40.0f / fchop_hz) * 1000.0f;
609 }
610 }
